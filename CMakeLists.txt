cmake_minimum_required(VERSION 3.22)
project(Harmonizer VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# JUCE
add_subdirectory(libs/JUCE)

# Rubber Band Library — single-file compilation (bundles KissFFT + Speex resampler)
add_library(rubberband STATIC
    libs/rubberband/single/RubberBandSingle.cpp
)
target_include_directories(rubberband PUBLIC libs/rubberband)
set_target_properties(rubberband PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(rubberband PRIVATE
    KISSFFT_DATATYPE=float
    NOMINMAX
)

# Plugin target
juce_add_plugin(Harmonizer
    COMPANY_NAME "Harmonizer"
    PLUGIN_MANUFACTURER_CODE Harm
    PLUGIN_CODE Hrm1
    FORMATS VST3 Standalone
    PRODUCT_NAME "Harmonizer"
    NEEDS_MIDI_INPUT TRUE
    IS_SYNTH FALSE
)

target_sources(Harmonizer PRIVATE
    src/PluginProcessor.cpp
    src/PluginEditor.cpp
    src/PitchDetector.cpp
    src/MidiNoteTracker.cpp
    src/HarmonyVoice.cpp
    src/VoicePool.cpp
    src/PanningEngine.cpp
)

target_include_directories(Harmonizer PRIVATE src)

target_link_libraries(Harmonizer
    PRIVATE
        juce::juce_audio_processors
        juce::juce_audio_utils
        juce::juce_dsp
        rubberband
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_warning_flags
)

target_compile_definitions(Harmonizer PUBLIC
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    $<$<PLATFORM_ID:Windows>:JUCE_ASIO=1>
)

# ============================================================================
# Test target — console app linking the same sources and JUCE modules
# ============================================================================

juce_add_console_app(HarmonizerTests
    PRODUCT_NAME "HarmonizerTests"
)

target_sources(HarmonizerTests PRIVATE
    # Test files
    tests/TestMain.cpp
    tests/TestMidiNoteTracker.cpp
    tests/TestPitchDetector.cpp
    tests/TestHarmonyVoice.cpp
    tests/TestVoicePool.cpp
    tests/TestPanningEngine.cpp
    tests/TestPluginProcessor.cpp
    # Plugin source files (shared with main target)
    src/PluginProcessor.cpp
    src/PluginEditor.cpp
    src/PitchDetector.cpp
    src/MidiNoteTracker.cpp
    src/HarmonyVoice.cpp
    src/VoicePool.cpp
    src/PanningEngine.cpp
)

target_include_directories(HarmonizerTests PRIVATE src tests)

target_link_libraries(HarmonizerTests
    PRIVATE
        juce::juce_audio_processors
        juce::juce_audio_utils
        juce::juce_dsp
        rubberband
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_warning_flags
)

target_compile_definitions(HarmonizerTests PRIVATE
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_STANDALONE_APPLICATION=1
    # Plugin macros normally generated by juce_add_plugin — define them
    # manually so the shared source files compile in the console app context.
    JucePlugin_Name="Harmonizer"
    JucePlugin_Manufacturer="Harmonizer"
    JucePlugin_ManufacturerCode=0x4861726d
    JucePlugin_PluginCode=0x48726d31
    JucePlugin_IsSynth=0
    JucePlugin_WantsMidiInput=1
    JucePlugin_ProducesMidiOutput=0
    JucePlugin_IsMidiEffect=0
)
